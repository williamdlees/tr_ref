Contigs were annotated with the reference set in ../ricotta_june_2 using the commands below.
link to annotation file:  http://vdjbase-distribution.s3-website-us-east-1.amazonaws.com/tcr_set/june_7/all_10x_projects_all_contigs.tsv.gz

docker run -it -v "%cd%":/data:z immcantation/suite:4.6.0 bash

export IGDATA=/usr/local/share/igblast
time igblastn \
   -germline_db_V ../ricotta_june_7/db/ricotta_TRV.fasta \
   -germline_db_J ../ricotta_june_7/db/ricotta_TRJ.fasta \
   -germline_db_D ../ricotta_june_7/db/ricotta_TRD.fasta \
   -query all_10x_projects_all_contigs.fasta \
   -auxiliary_data ../ricotta_june_7/ricotta_TR.aux \
   -custom_internal_data ../ricotta_june_7/ricotta_TR.ndm \
   -show_translation -outfmt 19 >all_10x_projects_all_contigs.tsv
